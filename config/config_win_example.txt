# config/config_win_example.txt
#
# copy this file to 'config_win.txt' in the same directory, then edit the paths.
#
# config/config_win.txt 被 .gitignore 忽略，每一台 windows 自己一份。
# 換一台 windows 就是重新複製一份、改下面幾行，go.bat 就能跑。
#
# 格式和 mac 的 config_vars.txt 一樣: KEY=VALUE
#   - # 開頭的整行是註解
#   - 行首不要有空白，= 的兩邊也不要有空白 (會變成變數名稱/值的一部分)
#   - 路徑不用加引號
#   - 結尾的反斜線加不加都可以，go.bat 會自己砍掉
#   - 這個檔案裡沒有機密。密碼在 config/config_secrets.txt，見最下面

# ---- go.bat 用的 -----------------------------------------------------------

# 來源: windows 本機的工作目錄。
#       camera_working 的正本在這裡，photoshop 的重活在這裡做。
#       這個目錄底下要有一個空的 it_exists.txt 當哨兵檔，沒有的話 go.bat 不跑。
local_camera_working_dir_base=C:\Users\jiech\Downloads\camera_working

# 目的: nas 上的 camera_working。
#       go.bat 只認這個路徑，路徑不在就直接離開，不會自己去連 nas。
#       坐在機器前面的話，檔案總管掛好的 T: 就是這個。
#       ssh 進來的話 T: 不存在，要先跑 mnt.bat，見下面。
#       也可以直接寫 UNC: \\DiskStation918\camera\camera_working
remote_camera_working_dir_base=T:\camera_working

# ---- mnt.bat 用的 (go.bat 不需要這三行) -------------------------------------
#
# 磁碟機代號是跟著 logon session 的。ssh 進來會拿到自己的 session，桌面上那個
# T: 在裡面不存在，所以 ssh 之後每次都要先跑 mnt.bat 把 T: 掛起來。
#
# 密碼不在這個檔案，在 config/config_secrets.txt 的 pw。那就是 mac 那邊
# '. go.sh 2' 在用的同一組 (DS918 的 admin)，不要再複製一份出來。
nas_drive=T:
nas_unc=\\DiskStation918\camera
nas_user=admin
